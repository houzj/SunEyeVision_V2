 [Fact]
 public void Configuration_check_reads_formal_runtime_and_graph_without_running()
 {
  var flow=new Flow("root","根");var tab=new WorkflowTabViewModel(RootCanvasFixture.Directory){Id=flow.Id};Bind(tab,flow);
  Assert.True(tab.IsRootConfigurationReady);Assert.Empty(flow.Nodes); // 空图就绪不代表可运行。
  var node=RootCanvasFixture.Node("node");flow.Nodes.Add(node);Assert.False(tab.IsRootConfigurationReady);Assert.Contains("没有可运行的图片",tab.RootConfigurationSummary);
  var policy=ImageSourcePolicy.Read(node)!;policy.FilePaths=new(){"image.png"};policy.ForRunPaths=new(){"image.png"};policy.CurrentIndex=0;policy.ApplyTo(node);typeof(WorkflowTabViewModel).GetMethod("RefreshRootReadiness",BindingFlags.Instance|BindingFlags.NonPublic)!.Invoke(tab,null);Assert.True(tab.IsRootConfigurationReady);
  flow.Connections.Add(new(){Id="bad",SourceNodeId="missing",TargetNodeId=node.Id});Assert.False(tab.IsRootConfigurationReady);
  flow.Connections.Clear();Assert.True(tab.IsRootConfigurationReady);
  node.ToolType="NotDeployed";Assert.False(tab.IsRootConfigurationReady);Assert.Contains("工具未加载",tab.RootConfigurationSummary);
  flow.Nodes.Clear();Assert.True(tab.IsRootConfigurationReady);
  var replacement=new Flow("other","替换");Bind(tab,replacement);flow.Nodes.Add(node);Assert.True(tab.IsRootConfigurationReady);Assert.Empty(replacement.Nodes);
 }
