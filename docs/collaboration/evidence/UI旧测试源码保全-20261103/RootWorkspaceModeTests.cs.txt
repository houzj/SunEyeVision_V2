using SunEyeVision.UI.Features.WorkflowUI.Canvas.Adapters;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Controls.Primitives;
using System.Windows.Media;
using System.Windows.Threading;
using SunEyeVision.UI.Shell.Controls.Shell;
using SunEyeVision.UI.Shell.Windows;
using Xunit;
namespace SunEyeVision.UI.Tests;

/// <summary>独立测试进程运行；真实WPF组装，不代表桌面鼠标或人工验收。</summary>
[Collection("VM canvas UI")]
public sealed class RootWorkspaceModeTests
{
 [Fact]
 public void Actual_mode_transition_closes_popups_hides_inapplicable_tabs_and_preserves_independent_scales()
 {
  Exception? failure=null;
  var thread=new Thread(()=>
  {
   Window? owner=null;
   try
   {
    if(Application.Current==null)_=new Application{ShutdownMode=ShutdownMode.OnExplicitShutdown};
    foreach(var path in new[]{"Shell/Resources/AppResources.xaml","Resources/Styles/CommonStyleResources.xaml","Themes/Dark/DarkTheme.xaml"})Application.Current.Resources.MergedDictionaries.Add(new(){Source=new Uri("/SunEyeVision.UI;component/"+path,UriKind.Relative)});
    ServiceInitializer.InitializeServices();var vm=new MainWindowViewModel();
    var shell=new ShellWorkspaceControl{DataContext=vm.WorkflowTabViewModel};shell.Initialize(vm);
    owner=new Window{Content=shell,Width=1100,Height=800,Left=-10000,ShowActivated=false};owner.Show();Pump();
    var tabs=(TabControl)shell.FindName("WorkflowTabControl");tabs.ApplyTemplate();
    var overview=(RootWorkflowOverviewControl)shell.FindName("RootOverview");
    var footer=(WorkspaceStatusBarControl)shell.FindName("WorkspaceStatusBar");
    var search=(WorkflowSearchControl)shell.FindName("WorkflowSearch");
    var locations=(WorkflowLocationControl)tabs.Template.FindName("WorkflowLocations",tabs);
    ((ToggleButton)search.FindName("Expand")).IsChecked=true;
    ((ToggleButton)locations.FindName("Expand")).IsChecked=true;
    vm.WorkflowTabViewModel.SelectedTab!.CurrentScale=1.3;
    shell.SetOverviewVisible(true);Pump();
    Assert.True(shell.IsRootOverviewVisible);
    Assert.False(((Popup)search.FindName("ResultsPopup")).IsOpen);Assert.False(((Popup)locations.FindName("LocationsPopup")).IsOpen);
    Assert.Equal(Visibility.Collapsed,((FrameworkElement)tabs.Template.FindName("TabPanelScrollViewer",tabs)).Visibility);
    Assert.Equal(Visibility.Collapsed,locations.Visibility);
    Assert.Equal(Visibility.Collapsed,((FrameworkElement)tabs.Template.FindName("PART_SelectedContentHost",tabs)).Visibility);
    var mode=(Button)tabs.Template.FindName("WorkflowViewButton",tabs);Assert.Equal("子流程",mode.ToolTip);
    overview.Viewport=new(1.7,40,60);Pump();
    var zoom=(Slider)footer.FindName("ZoomSlider");
    zoom.Value=2.5;Pump();Assert.Equal(2.5,overview.Viewport.Scale,6);
    zoom.Value=3.4;Pump();
    Assert.Equal(3.4,footer.Scale,6);Assert.Equal(1.3,vm.WorkflowTabViewModel.SelectedTab.CurrentScale,6);
    shell.SetOverviewVisible(false);Pump();
    Assert.False(shell.IsRootOverviewVisible);Assert.Equal(Visibility.Visible,locations.Visibility);
    Assert.Equal("全部流程",mode.ToolTip);Assert.Equal(1.3,footer.Scale,6);
    search.OpenIdSearch();Pump();Assert.Single(Application.Current.Windows.OfType<WorkflowIdSearchWindow>());
    shell.SetOverviewVisible(true);Pump();Assert.Empty(Application.Current.Windows.OfType<WorkflowIdSearchWindow>());
    Assert.Equal(3.4,overview.Viewport.Scale,6);Assert.Equal(3.4,footer.Scale,6);
    shell.SetOverviewVisible(false);Pump();
    vm.StatusBar.ShowExecutionSnapshotCommand.Execute(null);Pump();Assert.Single(Application.Current.Windows.OfType<WorkflowExecutionSnapshotWindow>());
    shell.SetOverviewVisible(true);Pump();Assert.Empty(Application.Current.Windows.OfType<WorkflowExecutionSnapshotWindow>());
    owner.Close();owner=null;Pump();
   }
   catch(Exception error){failure=error;}
   finally{owner?.Close();}
  });
  thread.SetApartmentState(ApartmentState.STA);thread.Start();Assert.True(thread.Join(TimeSpan.FromSeconds(45)));
  if(failure!=null)System.Runtime.ExceptionServices.ExceptionDispatchInfo.Capture(failure).Throw();
 }
 private static void Pump(){var frame=new DispatcherFrame();Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.Background,new Action(()=>frame.Continue=false));Dispatcher.PushFrame(frame);}
}
