    [Fact]
    public async Task Previous_real_invocation_cannot_replace_the_current_admitted_producer()
    {
        using var f=new Fixture();using var oldValues=new WorkflowGlobalRunValues(f.Solution.Id,f.Solution.GlobalVariables,Array.Empty<KeyValuePair<string,object?>>());
        using var oldRun=await f.Run(f.Request(f.SourceFlow,oldValues));Assert.True(oldRun.Success);
        var old=Assert.IsType<DeclaredOutputSnapshot>(oldRun.NodeResults["source"].Result);
        using var reader=oldRun.AcquireDeclaredOutput(oldRun.DeclaredOutputs.Single().Identity);
        f.Source.RuntimeConfiguration=f.Source.RuntimeConfiguration!.WithSettings(((Parameters)f.Source.RuntimeConfiguration.CaptureSettings()) with{Value=9});
        using var currentValues=new WorkflowGlobalRunValues(f.Solution.Id,f.Solution.GlobalVariables,Array.Empty<KeyValuePair<string,object?>>());var injected=false;
        void Inject(WorkflowNodeBase node,NodeExecutionResult completed)
        {
            if(node.Id!="source")return;Assert.Equal(9d,currentValues.Capture()[f.Variable.Name]);
            var property=typeof(AlgorithmNode).GetProperty(nameof(AlgorithmNode.LastToolResult))!;
            var context=typeof(AlgorithmNode).GetProperty("BindingContext",BindingFlags.Instance|BindingFlags.NonPublic)!.GetValue(node)!;
            var session=(WorkflowRunSession)context.GetType().GetField("_session",BindingFlags.Instance|BindingFlags.NonPublic)!.GetValue(context)!;
            var current=property.GetValue(node);
            try
            {
                property.SetValue(node,old);
                context.GetType().GetMethod("CompleteGlobalProducers",BindingFlags.Instance|BindingFlags.NonPublic)!.Invoke(context,new object[]{node.Id,session.RootContext!});
                injected=true;
            }
            finally{property.SetValue(node,current);}
        }
        f.Runner.NodeExecutionFinished+=Inject;
        try
        {
            using var run=await f.Run(f.Request(f.SourceFlow,currentValues));Assert.True(run.Success,string.Join(";",run.Errors));
            Assert.True(injected);Assert.False(currentValues.Capture().ContainsKey(f.Variable.Name));
            Assert.Equal("run-mismatch",currentValues.Diagnostics[f.Variable.Id].Code);
            Assert.Equal(9,Assert.IsType<Results>(run.NodeResults["source"].Result!.Output).Value);
            Assert.Equal(5d,reader.Values!.Get<double>("output.value"));
        }
        finally{f.Runner.NodeExecutionFinished-=Inject;}
    }

    public void Fragment_copy_remaps_module_indices_and_preserves_solution_variable_identities()
    {
        foreach(var variableSource in new[]{false,true})
        foreach(var variableIndex in new[]{false,true})
        {
            using var f=new Fixture(array:true);
            var indexNode=f.Add(f.SourceFlow,"index",true,1);
            var target=f.Add(f.SourceFlow,"target",false,0);
            var indexReference=new VariableReference(f.Solution.Id,"index-variable",DataType.Integer);
            f.Solution.GlobalVariables.Add(new GlobalVariable("index-value",1L,"Integer"){Id=indexReference.VariableId});
            BindingSelection selection=variableIndex?new BindingSelection.AtVariable(indexReference)
                :new BindingSelection.AtField(new OutputReference(indexNode.Id,"output.index",DataType.Integer,scopeId:f.SourceFlow.Id));
            if(variableSource)f.BindVariable(target,selection);
            else target.RuntimeConfiguration=target.RuntimeConfiguration!.WithFieldSettings(new Dictionary<string,ParameterSetting>{
                ["input"]=new ReferenceSetting(new OutputReference(f.Source.Id,"output.values",
                    GlobalProductionProbeDefinition.Outputs.Outputs.Single(p=>p.Id=="output.values").Type,
                    selection:selection,scopeId:f.SourceFlow.Id))});
            var before=JsonSerializer.Serialize(target.RuntimeConfiguration);
            var fragment=WorkflowFragmentCopy.Prepare(f.SourceFlow,new[]{f.Source.Id,indexNode.Id,target.Id},"copied-flow",
                id=>f.Solution.Workflows.SingleOrDefault(flow=>flow.Id==id),
                (node,_)=>new WorkflowNodeBase("copy-"+node.Id,node.Name,node.DispName,node.ToolType));
            var copied=fragment.Nodes.Single(node=>node.Id=="copy-target");
            var setting=copied.RuntimeConfiguration!.Fields.Read("input");
            BindingSelection copiedSelection;
            if(variableSource)
            {
                var source=Assert.IsType<VariableSetting>(setting).Reference;
                Assert.Equal(f.Solution.Id,source.SolutionId);Assert.Equal(f.Variable.Id,source.VariableId);
                copiedSelection=source.Selection;
            }
            else
            {
                var source=Assert.IsType<ReferenceSetting>(setting).Reference;
                Assert.Equal("copy-source",source.NodeId);Assert.Equal("copied-flow",source.ScopeId);
                copiedSelection=source.Selection;
            }
            if(variableIndex)
            {
                var index=Assert.IsType<BindingSelection.AtVariable>(copiedSelection).Reference;
                Assert.Equal(f.Solution.Id,index.SolutionId);Assert.Equal(indexReference.VariableId,index.VariableId);
            }
            else
            {
                var index=Assert.IsType<BindingSelection.AtField>(copiedSelection).Reference;
                Assert.Equal("copy-index",index.NodeId);Assert.Equal("copied-flow",index.ScopeId);
            }
            Assert.Equal(before,JsonSerializer.Serialize(target.RuntimeConfiguration));
            Assert.Equal(3,f.SourceFlow.Nodes.Count);
        }
    }
