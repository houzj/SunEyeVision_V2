using System;
using System.Linq;
using SunEyeVision.Plugin.SDK.Scripting;
using SunEyeVision.Plugin.SDK.Vision.Contracts;

public partial class UserScript : Script
{
    private int _invokeCount;

    public override void Init()
    {
        _invokeCount = 0;
        LogInfo("脚本初始化完成");
    }

    public override void Process()
    {
        _invokeCount++;

        // === 场景1: 输入图像信息（通用 IVisionImage 接口，不依赖具体引擎）===
        if (input1 == null || input1.IsEmpty)
        {
            LogError("输入图像 input1 为空，请在上游绑定图像源");
            return;
        }

        LogInfo($"[{_invokeCount}] 输入图像: {input1.Width}×{input1.Height}, 通道数={input1.Channels}");

        // === 场景2: 图像拷贝与像素数据（通用接口）===
        using (var clone = input1.Clone())
        {
            var pixelData = clone.GetPixelData();
            LogInfo($"图像拷贝像素数据: {pixelData.Length} 字节");
        }

        // === 场景3: 输出传递 ===
        output1 = input1;

        // === 场景4: 全局变量读写（通用接口优先）===
        // 4a. 用通用接口拿上游图像，不受引擎类型限制
        var upstream = GetGlobal<IVisionImage>("upstreamImage");
        if (upstream != null)
            LogInfo($"上游图像可用: {upstream.Width}×{upstream.Height}, 通道数={upstream.Channels}");

        // 4b. 推荐用通用接口类型 SetGlobal
        SetGlobal("lastImage", upstream);
        SetGlobal("lastWidth", input1.Width);
        SetGlobal("lastInvokeCount", _invokeCount);

        // === 场景5: 通用→算法类型转换（需在"程序集引用"中勾选对应引擎/适配层后取消注释使用）===
        // 方式一（推荐）：SDK 跨后端提取器 + 引擎类型。需勾选 halcondotnetxl 并 using HalconDotNet：
        //   // using SunEyeVision.Plugin.SDK.Vision.Hosting;
        //   // var hImage = VisionBackendExtractorLocator.ToHalcon?.Invoke(input1) as HImage;
        //   // if (hImage != null && hImage.IsInitialized()) { /* HOperatorSet.*/ }
        // 方式二：OpenCvSharp Mat。需勾选 OpenCvSharp 并 using OpenCvSharp：
        //   // var mat = VisionBackendExtractorLocator.ToOpenCv?.Invoke(input1) as Mat;
        // 方式三：适配层载体。需勾选 SunEyeVision.HalAdapter / SunEyeVision.CvAdapter 后：
        //   // if (input1 is SunEyeVision.HalAdapter.HalImage hal) { var h = hal.Unwrap(); }
    }
}
